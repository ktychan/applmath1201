\begin{frame}
  \frametitle{Mathematical models \notesref{Definition}{def:models}}

  A \hlmain{mathematical model} is one equation or a system of equations, typically with more than one variable, that describes the world around us.

  \bigskip
  \begin{itemize}
    \item<2-> Newton's second law \(F = ma\)
    \item<3-> The ideal gas law \(PV = nRT\)
    \item<4-> Exponential population growth \(P = P_{0} e^{rt}\)
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Discuss two models \notesref{Exercise}{ex:models-discussion}}

  Pick two or more models from the list and discuss with your study partner.

  \begin{enumerate}
    \item Do they match our everyday experience? Do they make sense?
    \item Are they accurate in every possible way? Do they have caveats and limitations?
    \item Are they built on certain assumptions? What happens when these assumptions break?
    \item Are they relatively easy to work with?
  \end{enumerate}
\end{frame}

% ---- lessons/intro-dimensional-homogeneity.tex
\begin{frame}
  \frametitle{Dimensional homogeneity \notesref{Definition}{def:dimensional-homogeneity}}

  A model is \hlmain{dimensionally homogeneous} if the units associated with each of its (additive) \emph{terms} match.

  A \emph{term} is a summand in the equation.

  \bigskip
  \onslide<2->{Also known as ``dimensionally consistent'' (often in physics).}
\end{frame}

\end{frame}

\begin{frame}
  \frametitle{Log-log transformation \notesref{Definition}{def:log-log-transformation}}

  The \hlmain{log-log transformation} of a power rule model \(y = c x^{\alpha}\) is
  \[
    \underbrace{\ln(y)}_{\text{v. axis}} = \overbrace{\alpha}^{\text{slope}} \underbrace{\ln(x)}_{\text{h. axis}} + \overbrace{\ln(c)}^{\text{v. intercept}}.
  \]

  Log-log transformations are ``undoable'' by raising both sides to base \(e\).
\end{frame}

  Suppose \(x\) and \(y\) are related by a power rule \(y = c x^{\alpha}\) with \(c > 0\).

\begin{frame}
  \frametitle{Recurrence \notesref{Definition}{def:recurrence}}

  In a \hlmain{discrete-time model} \(u_{t}\), time takes integer values. We call \(u_{t}\) the \hlmain{state at time \(t\)}.

  \medskip
  The model is described by a \hlmain{recurrence} when two things are specified.
  \begin{enumerate}
    \item<2-> An \hlmain{initial condition} \(u_{0}\).
    \item<3-> A \hlmain{recurrence equation} relating \(u_{t}\) to one or more \emph{previous} states.
  \end{enumerate}
\end{frame}

\begin{frame}
  \frametitle{Reading recurrences \notesref{Exercise}{ex:recurrence-identify}}

  For each recurrence below, identify its initial condition and its recurrence equation.

  \begin{align*}
    &p_{0} = 1 \quad\text{and}\quad p_{t} = \tfrac{1}{2} p_{t-1} - \tfrac{1}{2}
    &&a_{0} = 3 \quad\text{and}\quad a_{t} = a_{t-1} + 2 \\[1ex]
    &b_{0} = 4 \quad\text{and}\quad b_{t} = t \sqrt{b_{t-1}}
    &&f_{t} = \frac{1}{f_{t-1}} + 1 \\[1ex]
    &F_{t} = F_{t-1} + F_{t-2}
  \end{align*}
\end{frame}

\begin{frame}
  \frametitle{Solutions of a recurrence \notesref{Definition}{def:recurrence-solution}}

  A function \(f(t)\) is a \hlmain{solution} of a recurrence \(u_{t}\) with initial condition \(u_{0}\) if
  \begin{enumerate}
    \item<2-> \(f(t)\) satisfies the defining recurrence equation for \(u_{t}\), \emph{and}
    \item<3-> \(u_{0} = f(0)\).
  \end{enumerate}

  \bigskip
  \onslide<4->{\(f(t)\) cannot refer to any \(u_{t}\) and must be a well-defined function of \(t\).}
\end{frame}

\begin{frame}
  \frametitle{Verifying a proposed solution \notesref{Example}{ex:recurrence-verify-solution}}

  Is \(a_{t} = 2^{2^{t-1}}\) the solution for
  \[

  \]
\end{frame}

\begin{frame}
  \frametitle{Verifying a solution \notesref{Exercise}{ex:recurrence-verify-investment}}

  \[
    b_{t} = -1500 (1.02)^{t} + 2500
  \]

\end{frame}

\begin{frame}
  \frametitle{Solving \(u_{t} = a u_{t-1}\) \notesref{Method}{method:recurrence-exp}}

  Let \(a > 0\) be a constant. The solution to the recurrence \(u_{t} = a u_{t-1}\) is
  \[
    u_{t} = u_{0} a^{t},
  \]
  where \(u_{0}\) can be any constant.

  \bigskip
  \onslide<2->{None of the calculus tools apply \emph{directly} to a recurrence. \hlmain{Solving a recurrence converts an unfamiliar problem to a familiar problem.}}
\end{frame}

\begin{frame}
  \frametitle{Long-term behaviour \notesref{Example}{ex:recurrence-longterm-growth}}

  \begin{enumerate}
    \item \(r = 1\)?
    \item \(r > 1\)?
    \item \(0 < r < 1\)?
  \end{enumerate}
\end{frame}

\begin{frame}
  \frametitle{Equilibrium solutions \notesref{Definition}{def:recurrence-equilibrium}}

  An \hlmain{equilibrium solution} of a recurrence equation (not the whole recurrence) is a constant function. We drop the subscript and wear a hat: \(\hat{u}\).

  \bigskip
  \onslide<2->{Equilibria are easy to find. Replace every \(u_{t}\) by \(\hat{u}\) and solve for \(\hat{u}\) with basic algebra.}

  \medskip
  \onslide<3->{There can be more than one.}
\end{frame}

\begin{frame}
  \frametitle{The substitution method \notesref{Method}{method:recurrence-sub}}

  Consider \(v_{t} = A v_{t-1} + B\), whose equilibrium is always \(\hat{v} = \frac{B}{1 - A}\).

  \bigskip
  \begin{enumerate}
    \item<2-> Change variables: \(v_{t} = u_{t} + \hat{v}\).
    \item<3-> Solve the new recurrence \(u_{t}\).
    \item<4-> Recover the original solution.
  \end{enumerate}
\end{frame}

\begin{frame}
  \frametitle{Substitution, step by step \notesref{Example}{ex:recurrence-sub-worked}}

  Solve the recurrence \(b_{t} = 1.02 b_{t-1} - 50\) with initial condition \(b_{0} = 1000\).

  \begin{enumerate}[label={Step~(\arabic*).}]
    \item Perform the change of variable.
    \item Calculate \(u_{0}\) and solve for the new recurrence in \(u_{t}\).
    \item Recover the solution of the original recurrence.
  \end{enumerate}
\end{frame}

\begin{frame}
  \frametitle{Solution formula \notesref{Method}{method:recurrence-affine-formula}}

  The solution to the recursion \(v_{t} = A v_{t-1} + B\) with initial condition \(v_{0}\) is
  \[
    v_{t} = (v_{0} - \hat{v}) A^{t} + \hat{v}, \text{ where } \hat{v} = \frac{B}{1 - A}.
  \]
\end{frame}

\begin{frame}
  \frametitle{Formula and substitution \notesref{Exercise}{ex:recurrence-formula-vs-sub}}

  Solve the recurrence
  \[
    p_{0} = 1 \quad\text{and}\quad p_{t} = \frac{1}{2} p_{t-1} - \frac{1}{2}, \text{ for integers } t \ge 1
  \]
  using the formula \emph{and} the substitution method. Make sure you understand both.
\end{frame}

\end{frame}

\section{Week 3 --- Differential Equations}

% ---- lessons/diff-eq-intro.tex
\begin{frame}
  \frametitle{Birth and death rates \notesref{Example}{ex:diff-eq-intro}}

  Let \(N\) measure the size of a population, with time \(t\) flowing continuously and measured in years.

  \bigskip
  Assume the rate of change at time \(t\) is proportional to the size of the population at time \(t\). Express that rate as an equation if the per capita birth and death rates are \hlsupp{\(B = 10\%\)} and \hlsupp{\(D = 1\%\)} per year.

  \medskip
  Use dimensional homogeneity to stay on track.
\end{frame}

\begin{frame}
  \frametitle{Differential equations and IVPs \notesref{Definition}{def:differential-equation}}

  Let \(u\) be a \emph{differentiable} function of \(t\). A \hlmain{differential equation} (DE) relates \(u\) to one or more of its derivatives. The \hlsupp{unknown} is the function \(u\).

  \bigskip
  An \hlmain{initial-value problem} (IVP) has two parts:
  \begin{enumerate}
    \item<2-> An \hlmain{initial condition} --- some known value \(u(\text{constant})\).
    \item<3-> A differential equation.
  \end{enumerate}
\end{frame}

\begin{frame}
  \frametitle{Reading differential equations \notesref{Example}{ex:de-notation-examples}}

  \begin{itemize}
    \item With or without function notation:
      \(N'(t) = r N(t)\left(1 - \tfrac{N(t)}{K}\right)\) \; or \; \(N' = rN\left(1 - \tfrac{N}{K}\right)\).
    \item An IVP adds an initial condition: \(N(0) = 350\).
    \item The initial condition can be measured at any time: \(N(-3.236492) = 473829\).
    \item DEs can have higher-order derivatives:
      \(v = v' + \frac{v''}{t} + \frac{v^{(3)}}{t^{2}} + \sqrt{t-1}\).
  \end{itemize}
\end{frame}

% ---- lessons/diff-eq-solutions.tex
\begin{frame}
  \frametitle{Solutions of a DE \notesref{Definition}{def:de-solution}}

  Suppose we have a differential equation whose unknown is \(u(t)\).

  \bigskip
  A function \(f(t)\) is \hlmain{a solution of the differential equation} if we can \hlsupp{produce an equality} \emph{after} replacing \(u(t)\) in the DE with \(f(t)\).

  \bigskip
  \onslide<2->{Note the plural --- a DE generally has many solutions.}
\end{frame}

\begin{frame}
  \frametitle{Which functions solve \(u' = tu\)? \notesref{Example}{eg:de-solution-intro}}

  Which of the following are solutions to \(u'(t) = t u(t)\)?
  \[
    f_{1}(t) = e^{t^{2}}, \qquad f_{2}(t) = 2e^{t^{2}/2}, \qquad f_{3}(t) = 3e^{t^{2}/2}.
  \]
\end{frame}

\begin{frame}
  \frametitle{Prepare, calculate, decide \notesref{Method}{method:verify-de-solution}}

  \textbf{Prepare.}
  \begin{enumerate}
    \item Reorganize the DE to the form \hlmain{\((\cdots) = 0\)}.
    \item Rewrite \(u'(t)\) as \(\frac{d}{dt}\big( u(t) \big)\).
    \item Discard the ``\hlmain{\(= 0\)}'' on the right-hand side.
  \end{enumerate}

  \onslide<2->{\textbf{Calculate.} Replace every \(u(t)\) with \(f(t)\), then differentiate and simplify.}

  \medskip
  \onslide<3->{\textbf{Decide.} If you get \(0\), then \(f(t)\) is a solution; otherwise it is not.}
\end{frame}

\begin{frame}
  \frametitle{Which solves \(y' - 2y = 2t - 1\)? \notesref{Exercise}{ex:de-verify-linear}}

  Which of the following is a solution to \(\frac{dy}{dt} - 2y = 2t - 1\)?
  \[
    f_{1}(t) = 3e^{2t} - 1, \qquad f_{2}(t) = 3e^{2t} - 2.
  \]
\end{frame}

\begin{frame}
  \frametitle{Solutions of an IVP \notesref{Definition}{def:ivp-solution}}

  Suppose we have an initial-value problem whose initial condition is \(u(0)\).

  \bigskip
  A function \(f(t)\) is \emph{the} (one and only) solution to the IVP if
  \begin{itemize}
    \item<2-> \(f(0) = u(0)\), and
    \item<3-> \(f(t)\) is a solution to the differential equation.
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{The IVP sequel \notesref{Exercise}{eg:de-solution-intro-ivp}}

  Which of the following solves \(u'(t) = t u(t)\) with the initial condition \(u(0) = 2\)?
  \[
    f_{1}(t) = e^{t^{2}}, \qquad f_{2}(t) = 2e^{t^{2}/2}, \qquad f_{3}(t) = 3e^{t^{2}/2}.
  \]
\end{frame}

\begin{frame}
  \frametitle{Check the initial condition first \notesref{Exercise}{ex:de-ivp-check-first}}

  \[
    \sqrt{2x} - 1, \qquad \sqrt{2x - 1}, \qquad -\sqrt{2x + 1}.
  \]
\end{frame}

  \begin{enumerate}[label=(\alph*)]
    \item Find antiderivatives of \(x^{2}\).
    \item Integrate \(x^{2}\).
    \item Find the antiderivative \(F\) of \(f(x) = x^{2}\) satisfying \(F(0) = 4\).

\begin{frame}
  \frametitle{Equilibrium solutions \notesref{Definition}{def:diff-eq-equilibrium}}

  An \hlmain{equilibrium solution} (or equilibrium) of a differential equation is a constant function.

  \bigskip
  \onslide<2->{A common notation adds a bar to the dependent variable: \(\bar{u}\).}

  \medskip
  \onslide<3->{To find them, replace every \(u\) (or \(u(t)\)) by \(\bar{u}\) and solve. There may be none, one, or many.}
\end{frame}

\end{frame}

  \end{enumerate}
\end{frame}

\begin{frame}

  The \hlmain{order} of a differential equation is the order of the highest derivative of the dependent variable in the equation.

\begin{frame}

  Identify the order of the following differential equations.

    \qquad

  \]
\end{frame}

\begin{frame}
  \frametitle{Linear vs non-linear \notesref{Definition}{def:de-linear}}

  If a differential equation has no product of the dependent variable with itself or any of its derivatives, then it is called \hlmain{linear}; otherwise it is called \hlmain{non-linear}.
\end{frame}

\begin{frame}
  \frametitle{Find the linear DE \notesref{Example}{ex:de-identify-linear}}

    \qquad
    u'(t) + u'(t) u(t) = 0,
  \]
  \[
    \big(u(t)\big)^{2} + u'(t) u^{(5)}(t) = t u(t).
  \]
\end{frame}

\begin{frame}
  \frametitle{Linear, first-order DEs \notesref{Definition}{def:de-linear-first-order}}

  Assume \(u\) is a function of \(t\). A \hlmain{linear, first-order} differential equation has the form
  \[
    u'(t) + p(t) u(t) = q(t),
  \]
  where \(p(t)\) and \(q(t)\) can be any functions of \(t\).

  \bigskip
  \onslide<2->{The dependent variable \(u\) does not appear in \(p(t)\) or \(q(t)\).}
\end{frame}

  \onslide<2->{If an initial condition \(u(0)\) is given, the specific solution is \(u(t) = u(0) e^{kt}\).}
\end{frame}

\section{Week 5 --- Solving DEs, and Probability}

% ---- lessons/diff-eq-linear-first-order.tex
\begin{frame}
  \frametitle{The integrating factor formula \notesref{Theorem}{thm:diff-eq-linear-first-order}}

  To solve for \(u(t)\) in \(u'(t) + p(t) u(t) = q(t)\), an \hlmain{integrating factor} is
  \[
    v(t) = e^{\int p(t) \;dt} \quad (\text{choose } C = 0).
  \]

  \onslide<2->{The general solution is
  \[
    u(t) = \frac{1}{v(t)} \left( \int v(t) q(t) \;dt + C \right).
  \]}
\end{frame}

\begin{frame}
  \frametitle{Solving the green-tube IVP \notesref{Example}{ex:linear-first-order-green-tube}}

  Solve the initial-value problem
  \[
    G'(t) = 1 - 0.1\, G(t) \quad\text{and}\quad G(0) = 0
  \]
  from the green-liquid tube example.
\end{frame}

\begin{frame}
  \frametitle{Using an integrating factor \notesref{Method}{method:integrating-factor}}

  For \(u'(t) + p(t) u(t) = q(t)\), the integrating factor is \(v(t) = e^{\int p(t)\,dt}\).

  \begin{enumerate}
    \item<2-> Multiply the DE through by \(v(t)\).
    \item<3-> Reorganize the left-hand side as the result of the product rule.
    \item<4-> Integrate both sides, then isolate \(u(t)\).
  \end{enumerate}

  \bigskip
  \onslide<5->{This only applies to linear, first-order DEs.}
\end{frame}

\begin{frame}
  \frametitle{The Bernoulli substitution \notesref{Method}{method:bernoulli-sub}}

  The \hlmain{Bernoulli substitution} \(u = y^{1-n}\) applies to \hlsupp{non-linear}, first-order differential equations of the form

    y'(t) + P(t) y(t) = Q(t) {\color{warn} \big(y(t)\big)^{n}}.

\begin{frame}
  \frametitle{Mutually exclusive and exhaustive \notesref{Definition}{def:mutually-exclusive}}

  Two events \(E, F\) in \(\Omega\) are \hlmain{mutually exclusive} if \(E \cap F\) is empty.

  \bigskip
  \onslide<2->{Events \(E_{1}, \ldots, E_{n}\) are \hlmain{mutually exclusive and exhaustive} if \hlattn{every} outcome in \(\Omega\) belongs to \hlattn{exactly one} of them:
  \[
    \Omega = E_{1} \cup \cdots \cup E_{n}
    \quad\text{and}\quad
    E_{i} \cap E_{j} = \emptyset \text{ for } i \ne j.
  \]}
\end{frame}

\begin{frame}

  Let \(X\) be a discrete random variable.

  \bigskip
  \onslide<2->{The \hlmain{probability mass function} is a nonnegative \(f_{X}(x)\) with

    \item<2-> Expectation \(\mu = E(X) = \int_{-\infty}^{\infty} x f(x)\,dx\).

    \item<4-> Standard deviation \(\sigma = \sqrt{V(X)}\).
  \end{enumerate}
\end{frame}

\begin{frame}
  \frametitle{Summary statistics \notesref{Definition}{def:sample-statistics}}

  \hlmain{Sample data} is a list of numbers, typically measurements. Given \(x_{1}, \ldots, x_{n}\):

  \begin{enumerate}
    \item<2-> Sample mean \(\overline{X} = \frac{1}{n} \sum_{i=1}^{n} x_{i}\).
    \item<3-> Sample variance \(s^{2} = \frac{1}{n-1} \sum_{i=1}^{n} \left( x_{i} - \overline{X} \right)^{2}\).
    \item<4-> Sample standard deviation \(s = \sqrt{s^{2}}\).
  \end{enumerate}
\end{frame}

\begin{frame}
  \frametitle{Homogeneous systems \notesref{Method}{method:DE-homogeneous-linear-distinct-eigenvalues}}

  Find eigenvectors \(\vec{u}\) with eigenvalue \(\mu\) and \(\vec{v}\) with eigenvalue \(\lambda\), distinct.

  \bigskip
  \onslide<2->{The \hlmain{fundamental solution} for eigenpair \((\theta, \vec{w})\) is \(e^{\theta t}\vec{w}\), so the general solution is
  \[
    \vec{x}(t) = \alpha e^{\mu t} \vec{u} + \beta e^{\lambda t} \vec{v}.
  \]}

  \onslide<3->{For a specific solution, solve for \(\alpha, \beta\) using \(\vec{x}(0)\).}
\end{frame}

\begin{frame}

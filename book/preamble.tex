  keepparskip/.style={before upper pre={\setlength{\parskip}{0.5\baselineskip plus 2pt}}},
  every box/.style={keepparskip},
  admonition/.style={
    enhanced, breakable,
    boxrule=0.6pt, arc=2pt, left=6pt, right=6pt, top=4pt, bottom=4pt,
    fonttitle=\bfseries,
  },
}

% ------------------------------------------------------------
%           THEOREM-LIKE ENVIRONMENTS
% ------------------------------------------------------------
% One shared counter per chapter, as in the guided notes.
% Defined AFTER cleveref so that \cref says "Example", not "Statement".
\theoremstyle{definition}
\newtheorem{stmt}{Statement}[chapter]
\newtheorem{thm}[stmt]{Theorem}
\newtheorem{definition}[stmt]{Definition}
\newtheorem{example}[stmt]{Example}
\newtheorem{exercise}[stmt]{Exercise}
\tcolorboxenvironment{definition}{enhanced, breakable, blanker,
  borderline west={2pt}{0pt}{supp}, left=10pt, top=2pt, bottom=2pt}
\tcolorboxenvironment{example}{enhanced, breakable, blanker,
  borderline west={2pt}{0pt}{gray!60}, left=10pt, top=2pt, bottom=2pt}
\tcolorboxenvironment{exercise}{enhanced, breakable, blanker,
  borderline west={2pt}{0pt}{attn}, left=10pt, top=2pt, bottom=2pt}
% cleveref records labels of the shared counter as type `stmt`; alias the
% counter to the environment's own type while inside each environment.
\AtBeginEnvironment{thm}{\crefalias{stmt}{thm}}
\AtBeginEnvironment{definition}{\crefalias{stmt}{definition}}
\AtBeginEnvironment{example}{\crefalias{stmt}{example}}
\AtBeginEnvironment{exercise}{\crefalias{stmt}{exercise}}
\crefname{thm}{Theorem}{Theorems}
\crefname{definition}{Definition}{Definitions}
\crefname{example}{Example}{Examples}
\crefname{exercise}{Exercise}{Exercises}
\crefname{stmt}{Statement}{Statements}
